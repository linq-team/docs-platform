import { Badge } from '@stainless-api/docs/components';

## Plain badges

### Text only

{/** @type {const} */ (['sm', 'md', 'lg']).map((size) => (
  <div style={{ marginBottom: '1rem', display: 'flex', gap: '0.5rem' }} key={size}>
    {/** @type {const} */ (['none', 'info', 'success', 'warning', 'danger', 'note', 'tip', 'accent']).map((variant) => (
      <Badge
        key={variant}
        intent={variant}
        size={size}
      >
        {variant === 'none' ? 'Default' : variant.charAt(0).toUpperCase() + variant.slice(1)}
      </Badge>
    ))}
  </div>
))}

{/** @type {const} */ (['sm', 'md', 'lg']).map((size) => (
